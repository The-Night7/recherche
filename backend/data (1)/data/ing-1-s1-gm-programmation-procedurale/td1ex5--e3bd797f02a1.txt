#include<stdio.h>
int main(void)
{
    int num, den;
    printf(" veuillez saisir le numerateur et le denominateur");
    scanf("%d",&num);
    scanf("%d", &den);
    if( den !=0)
    {
        printf("%d / %d = %d", num, den, num/den);
    }
    else
    {
        printf ("division impossible");
    }
}
