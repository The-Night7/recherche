# Packages utiles
from math import *
import numpy as np

l1=np.array([1.0, 2.0, 3.0])
l2=np.array([2.0, 4.0, 6.0])

A = np.array([ l1, l2])

# print('A = ',A)


U, s, Vh = np.linalg.svd(A, full_matrices=True)
print('U=',U)
print('s=',s)
print('Vh=',Vh)

# # Dimension des matrices
n1,n2 =U.shape
m1,m2=Vh.shape
#
# # s est un vecteur, on reconstruit une matrice diagonale S
S = np.zeros((n1,m1))
S[:n1,:n1] = np.diag(s)
print('S=',S)
