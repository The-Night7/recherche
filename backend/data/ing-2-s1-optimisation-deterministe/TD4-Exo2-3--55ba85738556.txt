#test = f(x0 + rho * d0) - f(x0) + omega * rho * norm(d0) ** 2 > 0
def armijo(x,d,f):
    rho = 1
    test = f(x + rho * d) - f(x) + omega * rho * norm(d) ** 2 > 0
    while test:
        rho = tau * rho
        test = f(x + rho * d) - f(x) + omega * rho * norm(d) ** 2 > 0
    return rho

rho=armijo(x0,d0,f)

    g0 = gradf(x0)
    g1 = gradf(x1)
    beta = (norm(g1) / norm(g0)) **2
    d1 = - gradf(x1) + beta * d0
    rho = armijo(x1,d1,f)
    x2 = x1 + rho * d1
    test1 = (norm(x2 - x1) > eps * norm(x2))
    k += 1
    x0 = x1
    x1 = x2

    rho = armijo(x1,d1,f)
    x2 = x1 + rho * d1
    test1 = (norm(x2 - x1) > eps * norm(x2))

    x0 = x1
