double calcul_pi(int N){  
       double start,stop;  
       double cpu_time_used; 
       int i = 0;  
       double valeur_pi = 0.0;  
       double sommeFX = 0.0;  
       double X;  
       double h = 1.0 / (double)N; // pas d'integration
       start = 	omp_get_wtime();   
       for(i = 1; i < N; i++) {    
             X = h * (i - 0.5);    
             sommeFX += 4.0/(1.0 + X*X);  
       }  
       valeur_pi= h * sommeFX; 
       stop = omp_get_wtime();
       cpu_time_used = stop - start;		 
       printf("Valeur de pi %1.12f\n",valeur_pi);     
       return cpu_time_used;
}
