BROUILLON UNIX/SHELL - COMMANDES + THÉORIE
1. QUESTIONS DE COURS (THÉORIE)
 CHEMINS:
 ~ = home directory (/home/user)
 / = racine système
 . = répertoire courant
 .. = répertoire parent
 Chemin absolu: /home/user/file (depuis racine)
 Chemin relatif: ../file (depuis courant)
PIPE |:
 Redirige sortie cmd1 vers entrée cmd2
 Ex: ls | wc -l (compte fichiers)
 Communication inter-processus
FIND vs GREP:
 find: cherche NOMS de fichiers/répertoires
 grep: cherche CONTENU dans fichiers
MAN:
 man cmd: affiche manuel commande
 man -k mot: cherche dans manuels
 q pour quitter, / pour rechercher
RÉPERTOIRE:
 Dossier contenant liste/description fichiers
 Traité comme fichier spécial (type d)
SHELL:
 Interpréteur de commandes
 Affiche prompt, analyse, exécute commandes
 Crée processus
 echo $SHELL pour voir shell actuel
BASHRC:
 Fichier config shell (~/.bashrc)
 Lu à chaque démarrage terminal
 Contient alias, variables, config perso
 source ~/.bashrc pour recharger
ALIAS:
 Surnom pour commande longue
 alias ll='ls -la'
 unalias ll pour supprimer
 Définir dans .bashrc pour permanence
LIENS SYMBOLIQUES:
 Pointeur vers fichier/rép
 ln -s cible lien
 Si cible supprimée, lien cassé
 Économise espace disque

3. REDIRECTIONS & PIPES

cmd >> file            # Sortie→file (ajoute)
cmd < file             # Entrée←file
cmd 2> file            # Erreurs→file
cmd 2>> file           # Erreurs→file (ajoute)
cmd > file 2>&1        # Sortie+erreurs→file
cmd 2>/dev/null        # Ignore erreurs
cmd1 | cmd2            # Sortie cmd1→entrée cmd2
echo 'txt' > file      # Écrit texte

5. DROITS (r=4, w=2, x=1)
 chmod 755 file        # rwxr-xr-x
chmod 644 file        # rw-r--r--
chmod 777 file        # rwxrwxrwx
chmod 700 file        # rwx------
chmod 744 file        # rwxr--r--
chmod u+x file        # +exec proprio
chmod g+w file        # +write groupe
chmod o-r file        # -read autres
chmod -R 755 rep/     # Récursif
# Calcul: 7=rwx, 6=rw-, 5=r-x, 4=r--
# Ex: 754 = rwxr-xr--
# 1er chiffre: proprio
# 2e chiffre: groupe
# 3e chiffre: autres

15. TESTS & CONDITIONS
 # Syntaxe IF:
if [ condition ]; then
    commandes
elif [ autre_condition ]; then
    commandes
else
    commandes
fi
# IMPORTANT: espaces autour [ ]
# if [ $a -eq 5 ]; then  ← OK
# if [$a -eq 5]; then   ← ERREUR
# Tests ENTIERS:
[ $a -eq $b ]      # Égal (equal)
[ $a -ne $b ]      # Différent (not equal)
[ $a -lt $b ]      # < (less than)
[ $a -le $b ]      # ≤ (less or equal)
[ $a -gt $b ]      # > (greater than)
[ $a -ge $b ]      # ≥ (greater or equal)
# Tests CHAÎNES:
[ "$s1" = "$s2" ]  # Égal (TOUJOURS entre "")
[ "$s1" != "$s2" ] # Différent
[ -z "$s" ]        # Chaîne vide
[ -n "$s" ]        # Chaîne NON vide
# Tests FICHIERS:
[ -e file ]        # Existe
[ -f file ]        # Est fichier
[ -d rep ]         # Est répertoire
[ -r file ]        # Lisible (read)
[ -w file ]        # Modifiable (write)
[ -x file ]        # Exécutable
# Opérateurs LOGIQUES:
[ cond1 ] && [ cond2 ]  # ET logique
[ cond1 ] || [ cond2 ]  # OU logique
[ ! cond ]              # Négation (NON)
[ $a -eq 5 ] && [ $b -gt 3 ]  # Exemple
16. BOUCLES
 # Boucle FOR:
for i in 1 2 3 4 5; do
    echo $i
done
for i in {1..10}; do       # Séquence 1→10
    echo $i
done
for i in $(seq 1 10); do   # Avec seq
    echo $i
done
for file in *.txt; do      # Tous .txt
    echo $file
done
for i in $*; do            # Tous arguments
    echo $i
done
for i in $@; do            # Idem
    echo $i
done
# Boucle WHILE:
while [ condition ]; do
    commandes
done
i=1
while [ $i -le 10 ]; do
    echo $i
    i=$((i+1))
done
# Boucle UNTIL (inverse while):
until [ condition ]; do
    commandes
done
# Boucle INFINIE:
while true; do             # ou while [ 1 ]
    sleep 2
done
# BREAK & CONTINUE:
for i in {1..10}; do
    [ $i -eq 5 ] && continue  # Skip 5
    [ $i -eq 8 ] && break     # Stop à 8
    echo $i
done

18. EXEMPLES TYPES EXAM
 # Vérif nb arguments:
if [ $# -ne 1 ]; then
    echo "Usage: $0 <fic>"
    exit 1
fi
# Vérif si fichier:
if [ ! -f "$1" ]; then
    echo "Pas fichier"
    exit 2
fi
# Vérif si répertoire:
if [ ! -d "$1" ]; then
    echo "Pas rép"
    exit 2
fi
# Parcours fichiers rép:
for file in $1/*; do
    if [ -f $file ]; then
        echo $file
    fi
done
# Compte exécutables:
count=0
for file in $1/*; do
    if [ -x $file ] && [ -f $file ]; then
        count=$((count+1))
    fi
done
echo "Total: $count"
# Cherche & déplace:
mkdir -p Partiel
find . -name $1 -exec mv {} Partiel \;
# Tri fichier notes:
head -n 1 $1           # Entête
tail -n +2 $1 | sort   # Trie reste
# Menu interactif:
echo "1. Option 1"
echo "2. Quitter"
read -p "Choix: " c
case $c in
    1) date ;;
    2) exit 0 ;;
    *) echo "Err" ;;
esac
# Vérif login existe:
while true; do
    read -p "Login: " login
    [ "$login" = "quit" ] && break
    if id $login >/dev/null 2>&1; then
        groups $login
    else
        echo "Inexistant"
    fi
done
