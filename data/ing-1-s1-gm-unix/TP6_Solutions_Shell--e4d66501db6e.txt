3. Ajout des options (-h, -d, -v) avec getopts
 #!/bin/bash
verbose=0
rep="."
usage() {
    echo "Usage: $0 [-h] [-d repertoire] [-v]"
    exit 1
}
while getopts "hd:v" opt; do
    case $opt in
        h) usage ;;
        d) rep=$OPTARG ;;
        v) verbose=1 ;;
        *) usage ;;
    esac
done
for file in "$rep"/*.*; do
    ext="${file##*.}"
    [ $verbose -eq 1 ] && echo "Déplacement de $file vers $ext/"
    mkdir -p "$rep/$ext"
    mv "$file" "$rep/$ext/"
done

5. Vérifier un login et afficher ses groupes
 #!/bin/bash
while true; do
    read -p "Entrez un login (ou 'quit' pour arrêter): " login
    [ "$login" = "quit" ] && break
    if id "$login" >/dev/null 2>&1; then
        groups "$login"
    else
        echo "Login inexistant"
    fi
done

8. Appliquer un script à tous les sous-répertoires
 #!/bin/bash
script=$1
dir=$2
for sub in "$dir"/*; do
    [ -d "$sub" ] && $script "$sub"
done
