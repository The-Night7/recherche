#include <mpi.h>
#include <stdio.h>
int main(int argc , char ** argv)
{
   MPI_Init(&argc , &argv);
     int rank;
     int nb_procs;
	 
     MPI_Comm_rank (MPI_COMM_WORLD , &rank );
     MPI_Comm_size (MPI_COMM_WORLD , &nb_procs );
   
     printf (" Je suis le processus %d parmi %d", rank, nb_procs );
   
   MPI_Finalize();
 return 0;
}
