function loadDoc() { 
  var xhttp = new XMLHttpRequest(); 
  xhttp.onreadystatechange = function() { 
    if (this.readyState == 4 && this.status == 200) { 
    var reponse = JSON.parse(xhttp.responseText); 
    var txt=""; 
    for( var i in reponse.produits) 
    { txt=txt+reponse.produits[i].nom;
