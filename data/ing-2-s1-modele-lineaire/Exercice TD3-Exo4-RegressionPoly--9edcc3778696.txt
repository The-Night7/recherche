    OU BIEN
# Coefficients du modèle
R2 <- 0.9 # Coefficient de détermination
p <- 2    # Nombre de prédicteurs
n <- 11   # Taille de l'échantillon

# Calcul de la statistique de Fisher
F_stat <- (R2 / p) / ((1 - R2) / (n - p - 1))

# Degrés de liberté
num_df <- p
denom_df <- n - p - 1

# Calcul de la p-valeur
p_value <- 1 - pf(F_stat, df1 = num_df, df2 = denom_df)

# Affichage de la p-valeur
cat("Statistique de Fisher (F) :", F_stat, "\n")
cat("Degrés de liberté (numérateur, dénominateur) :", num_df, denom_df, "\n")
cat("P-valeur :", p_value, "\n")

## On trouve $p=10^{-4}$ c'est la même  que summary(poly_model) nous a donné.

7. Déterminer la p-valeur du test de nullité de a2

# Coefficients du modèle
p <- 2    # Nombre de prédicteurs
n <- 11   # Taille de l'échantillon

# Estimation de a2 et de la variance de l'estimation
a2_hat <- 0.4663
ecartype_a2_hat <- 0.0603

# Calcul de la statistique de Student
T_Student <- a2_hat / ecartype_a2_hat

# Degrés de liberté
num_df <- 1  # Degré de liberté pour le numérateur
denom_df <- n - p - 1  # Degré de liberté pour le dénominateur

# Calcul de la p-valeur (test bilatéral)
p_value <- 2 * pt(-abs(T_Student), df = denom_df)  #la fonction pt()  est utilisée pour calculer
                                                   # la fonction de distribution cumulative d'une 
                                                   # distribution de Student t
